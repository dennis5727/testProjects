#LIKE % _       # matching(case insensitive, space is also counted), = is case sensitive
SELECT "title" FROM "longlist"
WHERE "title LIKE '%love%';             # % is anything

SELECT "title" FROM "longlist"
WHERE "title LIKE 'The%';               # books that start with 'The' or even 'Their'

SELECT "title" FROM "longlist"
WHERE "title LIKE '%love%'; 

SELECT "title", "author" FROM "longlist"
WHERE "title LIKE 'P_re';               # _ is single character


# > < >= <=        for dealing with range
SELECT "tilte", "author", "year" FROM "longlist"
WHERE "year" >= 2019 AND "year" <= 2023;
# BETWEEN ... AND ...        both endpoints inclusive
SELECT "tilte", "author", "year" FROM "longlist"
WHERE "year" BETWEEN 2019 AND 2022;

SELECT "tilte", "author", "rating", "votes" FROM "longlist"
WHERE "rating" > 4.0 AND "votes" >= 10000;


# ORDER BY, ASC, DESC                  # default order is ASC
SELECT "title", "rating", "votes" FROM "longlist"
ORDER BY "rating" DESC, "votes" DESC                # first compare the rating,
LIMIT 10;                                           # votes is the tie-breaker
